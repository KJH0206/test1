#include <Servo.h>
Servo myservo;
int const photocell = A0;
int photocellVal ;
int angle;
int sensorMax = 0 ;
int sensorMin = 1023 ;

void setup () {
myservo.attach(9);
Serial.begin(9600);
while (millis() < 5000) {
  int sensorValue = analogRead(photocell);
if (sensorValue > sensorMax) {      //큰 값이 들어오면 바뀜
    sensorMax = sensorValue;
    }
if (sensorValue < sensorMin) {  //작은 값이 들어오면 바뀜
    sensorMin = sensorValue;
    }
  }
}

void loop() {
  photocellVal = analogRead(photocell);   포트셀 일종의 저항, 저항을 연결하여 그 깍아먹은 저항의 비만큼 전류
  Serial.print("photocellVal: ");
  Serial.print(photocellVal);
  angle = map(photocellVal, sensorMin, sensorMax, 0, 179);
  Serial.print(", angle: ");
  Serial.println(angle);
  myservo.write(angle);
  delay(15);
}

융합문제 1

int ledPin = 9;
int fadeAmount ;
int brightness = 0;
void setup() {
}
void loop() {
  int potVal = analogRead(A0);
  fadeAmount = map(potVal, 0 , 1023, 0 , 10);
if (fadeAmount==0){
    fadeAmount = 1;
  }
  for (brightness = 0 ; brightness <= 255; brightness += fadeAmount) {
analogWrite(ledPin, brightness);
delay(50);
  }
  for (brightness = 255 ; brightness >= 0; brightness -= fadeAmount) {
analogWrite(ledPin, brightness);
delay(50);
  }
}

 }
